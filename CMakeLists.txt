cmake_minimum_required(VERSION 3.20)
project(cuda_particle_simulator LANGUAGES CXX)
option(ENABLE_CUDA "Build CUDA linked-cell backend and GPU tests" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(particles_cpu src/cpu.cpp)
target_include_directories(particles_cpu PUBLIC include)
if(MSVC)
  target_compile_options(particles_cpu PRIVATE /W4)
else()
  target_compile_options(particles_cpu PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(particle_sim src/main.cpp)
target_link_libraries(particle_sim PRIVATE particles_cpu)
enable_testing()
add_executable(cpu_test tests/cpu_test.cpp)
target_link_libraries(cpu_test PRIVATE particles_cpu)
add_test(NAME cpu_reference COMMAND cpu_test)
add_test(NAME cli_smoke COMMAND particle_sim --count 17 --steps 3)
add_test(NAME cli_reject_invalid COMMAND particle_sim --count 0)
set_tests_properties(cli_reject_invalid PROPERTIES WILL_FAIL TRUE)

if(ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75)
  endif()
  enable_language(CUDA)
  add_library(particles_cuda src/cuda.cu)
  target_include_directories(particles_cuda PUBLIC include)
  target_link_libraries(particles_cuda PUBLIC particles_cpu)
  set_target_properties(particles_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_definitions(particle_sim PRIVATE HAS_CUDA)
  target_link_libraries(particle_sim PRIVATE particles_cuda)
  add_executable(gpu_test tests/gpu_test.cpp)
  target_link_libraries(gpu_test PRIVATE particles_cuda)
  add_test(NAME gpu_reference COMMAND gpu_test)
  set_tests_properties(gpu_reference PROPERTIES SKIP_RETURN_CODE 77)
endif()
